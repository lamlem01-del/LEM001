## S17 — ផែនដីព្រះបាទពញាយ៉ាត (ពីគ.ស.១៣៥៦ ដល់ គ.ស.១៣៥៩)
**ស្ថានភាពអាន៖** បានអានគ្រប់កថាខណ្ឌ ¶1–¶12

### សង្ខេប
ព្រះពញាយ៉ាតឡើងរាជ្យនៅព្រះជន្ម ៥១ ហើយឯកសារសរសើរថាទ្រង់កាន់ទសពិធរាជធម៌ (¶1)។ ក្រោយមក ទ្រង់ប្រជុំព្រះរាជវង្ស និងមន្ត្រីទាំងអស់ ហើយត្រាស់ពន្យល់ហេតុផលនៃការផ្លាស់រាជធានី។ ហេតុផលទាំងនោះគឺ៖ សៀមជាសត្រូវតាំងពីដើម ហើយបានកៀរគ្រួខ្មែរយកទៅច្រើន, ប្រជាជនក្នុងក្រុងនៅសល់តិច, បន្ទាយសៀមរឹងមាំ, ទាហាននឿយហត់, ស្បៀងមិនគ្រប់គ្រាន់, និងគ្មានខេត្តរបាំងខាងក្រៅ។ ទ្រង់ស្នើឲ្យថយទៅតាំងនៅបាសាន (¶2–¶3)។ ទ្រង់សួរយោបល់ ហើយមន្ត្រីយល់ព្រម (¶4)។ ទ្រង់កេណ្ឌពលជើងទឹកជើងគោក ហើយយាងដោយក្បួនទូកតាមឫក្សពេលល្អ ទៅតាំងព្រះរាជវាំងបែរមុខទៅកើតនៅបាសាន (¶5)។ ក្រុងថ្មីនោះមានឈ្មោះ «ព្រះវាំងថ្ម» (¶6)។

បន្ទាប់មក អ្នករៀបរៀងប្តូរទៅរឿងទឹកជំនន់នៅប្រទេសលាវ។ ដើមគគីរមួយដែលមានព្រះពុទ្ធរូបពីរអង្គ រសាត់មកដល់ទីនេះ។ ដូនពេញ ជាឧបាសិកាអ្នកមាន ព្យាយាមលើកព្រះនោះ តែលើកមិនរួច ទើបពឹងព្រះធម្មលង្កា។ លោកធ្វើពិធី ៧ ថ្ងៃ រួចប្រគល់ព្រះមួយអង្គឲ្យដូនពេញ ហើយទុកមួយអង្គនៅព្រែកបង្កង (¶7)។ ដូនពេញប្រមូលអ្នកស្រុកលើកភ្នំ ដើម្បីតម្កល់ព្រះនោះ (¶8)។

ទឹកលិចរាជធានីបាសាន ហើយមន្ត្រីលំបាកណាស់។ ពេលទឹកស្រក ព្រះរាជាផ្លាស់មកឫស្សីកែវម្តងទៀត។ ទ្រង់បង្គាប់ចៅពញាតេជោឲ្យលើកដីចាក់បំពេញ (ទើបមានឈ្មោះ «បឹងតេជោ») ហើយបង្គាប់ផ្លុងឲ្យជីកព្រែក (ទើបមានឈ្មោះ «ព្រែកពាមផ្លុង»)។ ទ្រង់ក៏សាងបន្ទាយ អូរ និងកំពែងផងដែរ (¶9)។ ទ្រង់សាងវត្ត និងខ្លោងមឿង «អ្នកតាប្រាជ្ញ» ហើយដាក់ឈ្មោះក្រុងថា «ចតុមុខ»។ ឯកសារពិពណ៌នាថាក្រុងនោះរុងរឿង ហើយបរទេសនាំបណ្ណាការមកថ្វាយ (¶10)។ ទ្រង់មានព្រះរាជបុត្របីអង្គ ដែលកើតពីព្រះមាតាបីផ្សេងគ្នា ហើយទ្រង់ប្រទានយសស្មើៗគ្នា (¶11)។ ទ្រង់សុគតនៅព្រះជន្ម ៧៨។ មន្ត្រីអញ្ជើញព្រះនារាយណ៍រាជា ដែលជាបុត្រច្បង ឲ្យឡើងស្នងរាជ្យ (¶12)។

- ¶2 ឆ្នាំថោះ នពស័ក ព.ស. ១៩៣១ - គ.ស. ១៣៨៧ - ម.ស. ១៣០៩ - ច.ស. ៧៤៩ («គ្រងរាជ្យ ២ ឆ្នាំ សោយរាជ្យ ៥ ឆ្នាំ ត្រូវជា ៧ ឆ្នាំ»)
- ¶5 ចេញដំណើរ៖ «ឆ្នាំថោះ នពស័ក ព.ស. ១៩២៦ គ.ស. ១៣៨២»
- ¶7 ទឹកជំនន់នៅលាវ៖ ឆ្នាំមមី សំរិទ្ធិស័ក ព.ស. ១៩២២ - គ.ស. ១៣៧៨ - ម.ស. ១៣០០ - ច.ស. ៧៤០
- ¶12 រយៈរាជ្យ៖ ធ្វើសង្គ្រាម ១ ឆ្នាំ + មហានគរ ១៥ + បាសាន ១ + ក្រុងថ្មី ៣០ = ៤៧ ឆ្នាំ; ព្រះជន្ម ៧៨

### ការប្រឹក្សា ទំនាយ សាសនា ពិធី
- ¶4 ការប្រឹក្សាជាផ្លូវការ៖ ព្រះរាជាស្នើ ហើយមន្ត្រីយល់ព្រមជាឯកច្ឆន្ទ (គ្មានសំឡេងជំទាស់)។
- ¶5 ការជ្រើសរើសឫក្សមង្គល៖ ព្រះហោរាធិបតីគោះគង ហើយមានស្វេតច្ឆ័ត្រ។ អ្នករៀបរៀងពន្យល់ពាក្យ «អភិរម»។
- ¶7 ព្រះធម្មលង្កាធ្វើពិធី ៧ ថ្ងៃ ដោយប្រើសរសៃអំបោះ ដើម្បីយកព្រះពុទ្ធរូប។
- ¶10 ការសាងខ្លោងមឿង (អ្នកតាប្រាជ្ញ) នៅទិសអគ្នេយ៍នៃវាំង ក្បែរវត្តពាមផ្លុង។ ព្រះពុទ្ធសាសនារុងរឿងដោយសារ «ព្រះបារមី» និងទសពិធរាជធម៌។
- ¶12 ទុកព្រះសពក្នុងព្រះកោដ្ឋ មុនអញ្ជើញអ្នកស្នងរាជ្យ។

### តួលេខ (កងទ័ព ដំរី ឆ្នាំ អាយុ)
- ព្រះជន្ម ៥១ ពេលឡើងរាជ្យ (¶1); ព្រះជន្ម ៧៨ ពេលសុគត (¶12)
- ដំណាក់កាលរាជ្យ ១+១៥+១+៣០ = ៤៧ ឆ្នាំ (¶12); ៧ ឆ្នាំមុនផ្លាស់រាជធានី (¶2)
- ពិធី ៧ ថ្ងៃ និងព្រះរសាត់ ៧ ថ្ងៃ (¶7)
- ព្រះរាជបុត្រ ៣ អង្គ (¶11)
- មិនមានតួលេខកងទ័ពទេ

### ភាសាវាយតម្លៃ (សរសើរ/មើលងាយ ឧ. «អា…»)
- សរសើរ៖ «ទសពិធរាជធម៌ ទំនុកបម្រុងព្រះពុទ្ធសាសនាណាស់» (S17¶1); «ព្រះមហាសុចរិត ទសពិធរាជធម៌ មហាប្រសើរ» (S17¶10); សត្រូវ «កោតក្រែងស្ញប់ស្ញែងខ្លាចព្រះបារមី» (S17¶10)។
- ព្រះរាជាប្រើពាក្យ «វា» ចំពោះសៀម៖ «វាផាត់កៀរគ្រួយកអស់» «បើសៀមវាលើកមក» (S17¶3)។
- មិនឃើញពាក្យ «អា» ទេ។

### សំឡេងអ្នករៀបរៀង និងប្រភពដែលដកស្រង់ (ពង្សាវតារ សាស្ត្រា ការពន្យល់ពាក្យ ការនិយាយទៅអ្នកអាន)
- ការពន្យល់ពាក្យ៖ «អភិរម ឬ អភិរំ = ឆត្ររួត…» (¶5); «ខ្លោងមឿង គឺអ្នកតាចាំស្រុក… ឃ្លាំងមឿង» (¶10)។
- ការភ្ជាប់ទីកន្លែងបុរាណទៅបច្ចុប្បន្ន៖ «ព្រែកពាក់ព្រ័ត្រ គឺព្រែកលាភសព្វថ្ងៃនេះ» និង «កោះឫស្សីកែវ (គឺក្រុងភ្នំពេញយើងសព្វថ្ងៃនេះ)» (¶7)។
- ការប្រៀបធៀបកំណែ៖ «តាមព្រះរាជពង្សាវតារ ដែលតម្កល់ទុក នឹង ព្រះរាជបណ្ណាល័យក្រុងភ្នំពេញ» ថាឧកញ៉ាផ្លុងជាចៅហ្វាយស្រុកបាទី (¶9)។
- ការពន្យល់ប្រភពដើមឈ្មោះទីកន្លែង (បឹងតេជោ ព្រែកពាមផ្លុង) (¶9)។
- ¶7 ចាប់ផ្តើមដោយ «នេះនឹងនិយាយពីក្រុងលាវវិញ» ដែលជាឃ្លាប្តូររឿងតាមរបៀបពង្សាវតារ។

### ស្ត្រី ប្រជាជនសាមញ្ញ ជនជាតិ/បរទេស
- **ស្ត្រី៖** ដូនពេញជាអ្នកផ្តួចផ្តើម និងដឹកនាំសហគមន៍ក្រៅរដ្ឋ (¶7–¶8)។ ព្រះមាតាបីអង្គមានឈ្មោះ (¶11)។ ក្នុងក្បួនហែមាន «ស្នំ ព្រហ្មចារី… ទូកជំទាវ នាងខុនណាង» (¶5)។
- **ប្រជាជន៖** ត្រូវកេណ្ឌជា «ពល កសិករ… ទាស កម្មករ» ដើម្បីផ្លាស់ទីលំនៅ (¶5)។ រាស្ត្រលើកដីបង្កើតបឹងតេជោ (¶9)។ រាស្ត្រមកជួយដូនពេញដោយស្ម័គ្រចិត្ត «ជាអចិន្ត្រៃយ៍» (¶8)។ ការលំបាកនៃទឹកជំនន់ត្រូវរៀបរាប់ថាជាការលំបាករបស់ខ្ញុំរាជការ និងព្រះរាជវង្ស (¶9)។
- **បរទេស៖** សៀមជាសត្រូវ (¶3)។ ព្រះពុទ្ធរូបមកពីលាវ (¶7)។ «ចិន យួន ចាម ជ្វា» នាំបណ្ណាការមកថ្វាយ (¶10)។ មាន «ឈ្មួញជួញប្រែក្នុងក្រៅប្រទេស» (¶10)។

### កំហុសអត្ថបទ និងភាពមិនស៊ីគ្នា
- ចំណងជើងដាក់ គ.ស. ១៣៥៦–១៣៥៩ តែ ¶1 ដាក់ការឡើងរាជ្យនៅ គ.ស. ១៣៨២ ហើយ ¶12 និយាយពីរាជ្យ ៤៧ ឆ្នាំ។ ចំណងជើងនេះទំនងជាចម្លងមកពី S11 (L013) (ស្របនឹងកំណត់សម្គាល់ក្នុងបញ្ជី L020)។
- ¶1 ហៅ គ.ស. ១៣៨២ ថា «ឆ្នាំច» តែ ¶5 ហៅ គ.ស. ១៣៨២ ថា «ឆ្នាំថោះ នពស័ក»។ ¶2 ក៏ហៅ គ.ស. ១៣៨៧ ថា «ឆ្នាំថោះ នពស័ក» ដែរ។ ដូច្នេះកាលបរិច្ឆេទចេញដំណើរនៅ ¶5 ផ្ទុយនឹង ¶2 (ក្រោយរាជ្យ ៧ ឆ្នាំ)។
- ¶7 ទឹកជំនន់នៅលាវ គ.ស. ១៣៧៨ មុនការឡើងរាជ្យ (១៣៨២) ទោះបីរឿងនេះត្រូវបានដាក់ក្នុងរាជ្យនេះ។
- ¶1 ព្រះជន្ម ៥១ + រាជ្យ ៤៧ ឆ្នាំ = ៩៨ ឆ្នាំ មិនមែន ៧៨ (¶12)។ ¶2 ថាគ្រងរាជ្យ ២ ឆ្នាំ + សោយរាជ្យ ៥ ឆ្នាំ ក៏មិនស៊ីនឹង ¶12 (ធ្វើសង្គ្រាម ១ ឆ្នាំ + មហានគរ ១៥ ឆ្នាំ) ដែរ។
- ¶6 ក្រុងនៅបាសានមានឈ្មោះ «ព្រះវាំងថ្ម» ហើយ ¶9 ហៅថា «ព្រះរាជវាំងថ្ម»។
- ¶7 ព្រះពុទ្ធរូប «ពីរព្រះអង្គ» ដែល «មានព្រះភក្ដ្រឈមជាបួនទិស»។ ចំនួន និងរូបរាងមិនច្បាស់។
- ¶4 មានសញ្ញា «\» ដែលជាសំណល់ទម្រង់អត្ថបទ។

### ការភ្ជាប់ទៅករណី (C/K) និងករណីថ្មីដែលស្នើ

**ស្ថានភាពអាន៖** បានអានគ្រប់កថាខណ្ឌ ¶1–¶6

### សង្ខេប
ព្រះនារាយណ៍រាជាឡើងរាជ្យនៅព្រះជន្ម ៣៣ ដោយមានព្រះនាមរាជាភិសេកវែង។ ទ្រង់សាងព្រះមេរុធំមានកំពូលបួន ហើយធ្វើបុណ្យព្រះបរមសពព្រះបិតា ៧ ថ្ងៃ។ ព្រះមាតា សម្ដេចព្រះមាលាទេវី សាងវាំងនៅព្រះវិហារឪផ្រឹង (¶1)។ ពេលយាងទៅគាល់ព្រះមាតា ព្រះរាជាទតឃើញដូនពេញ និងបក្សពួក «ជាងរយពាន់នាក់» កំពុងជញ្ជូនដី ទើបទ្រង់សួរ (¶2)។ មន្ត្រីទូលពន្យល់ថា ដូនពេញកំពុងកៀរគរអ្នកស្រុកលើកភ្នំ ដើម្បីតម្កល់ព្រះចតុម្មុខ (¶3)។ ព្រះរាជាគិតថា ដូនពេញ «ជារាស្ត្រពិត» តែកុសលរបស់គាត់ល្អ ហើយក្សត្រគួរចូលជួយ។ ទ្រង់នាំព្រះមហេសី ស្នំ និងមន្ត្រីទៅជញ្ជូនដីដោយផ្ទាល់។ ឃើញដូច្នេះ រាស្ត្រទាំងក្នុងក្រៅស្រុកក៏ចូលរួមដែរ រហូតបានជាភ្នំ។ ទ្រង់សាងចេតិយ បញ្ចុះព្រះបរមសារីរិកធាតុ ហើយកាត់ដើមគគីរជាពីរ ដើម្បីធ្វើរូបតំណាងព្រះចតុម្មុខ។ ទ្រង់បណ្តាំថា បើមានសត្រូវ ត្រូវកប់ព្រះពុទ្ធរូបដើមកុំឲ្យគេយកបាន។ ទ្រង់ដាក់ឈ្មោះ «វត្តភ្នំពេញ» ហើយប្តូរឈ្មោះខេត្តកោះឫស្សីកែវទៅជា «ខេត្តភ្នំពេញ» (¶4)។ នៅ គ.ស. ១៤៣២ ទ្រង់សាងចេតិយនៅចង្កេះភ្នំពេញ ដើម្បីបញ្ចុះព្រះបរមធាតុព្រះបិតា។ ទ្រង់មានព្រះរាជបុត្រមួយអង្គ គឺព្រះសុរិយោទ័យ (¶5)។ ទ្រង់សុគតនៅព្រះជន្ម ៣៩ បន្ទាប់ពីរាជ្យ ៦ ឆ្នាំ។ មន្ត្រីអញ្ជើញព្រះអនុជ (ព្រះសេរីរាជា) ឲ្យឡើងស្នង មិនមែនព្រះរាជបុត្រទេ។ ឯកសារបន្ថែមលេខយោងពីសាស្ត្រាវត្តកំពង់ត្រឡាចក្រោម អំពីប្រភពនៃរាជពង្សាវតារ (¶6)។

### កាលបរិច្ឆេទ (ដូចសរសេរក្នុងអត្ថបទ)
- ¶1 ឡើងរាជ្យ៖ ឆ្នាំរោង ទោស័ក ព.ស.១៩៤៤-គ.ស. ១៤២៧-ម.ស. ១៣៤៩-ច.ស. ៧៨៩ (ព្រះជន្ម ៣៣)
- ¶5 ឆ្នាំជូត ចត្វាស័ក ព.ស. ១៩៧៦-គ.ស. ១៤៣២-ម.ស. ១៣៥៤-ច.ស. ៧៩៤ (សាងចេតិយ)
- ¶6 សោយរាជ្យ ៦ ព្រះវស្សា, ព្រះជន្ម ៣៩

### ការប្រឹក្សា ទំនាយ សាសនា ពិធី
- ¶1 ព្រះមេរុ (មានវិមាត្រ) និងបុណ្យ ៧ ថ្ងៃ; ¶5 បុណ្យសង្គាយនា ៧ ថ្ងៃ។
- ¶4 ព្រះពុទ្ធរូបជាវត្ថុការពាររាជធានី («មិនអោយរាជសត្រូវ… យកព្រះនគរបាន») ហើយមានបណ្តាំឲ្យកប់ព្រះពុទ្ធរូបពេលមានសត្រូវ។ «ការបកស្រាយ»៖ វត្ថុសក្ការៈត្រូវបានប្រើជាឧបករណ៍សន្តិសុខរាជធានី។
- ¶4 ការបញ្ចុះព្រះបរមសារីរិកធាតុក្នុងចេតិយ។
- ¶2–¶3 ការសួររបស់ព្រះរាជា និងការទូលរបស់មន្ត្រី ជាការប្រឹក្សាតូចមួយ។

### តួលេខ (កងទ័ព ដំរី ឆ្នាំ អាយុ)
- ព្រះមេរុ៖ កំពូល ៤ កំពស់ «៣ សិន ១០ ព្យាម ១ ហត្ថ ១ ចំអាម» (¶1)
- អ្នកជញ្ជូនដី «ជាងរយពាន់នាក់» (¶2)
- ព្រះជន្ម ៣៣ (¶1) និង ៣៩ (¶6), រាជ្យ ៦ ឆ្នាំ (¶6)
- បុណ្យ ៧ ថ្ងៃ (¶1, ¶5)

### ភាសាវាយតម្លៃ (សរសើរ/មើលងាយ ឧ. «អា…»)
- ព្រះរាជាគិតថា៖ «ដូនពេញនេះ ជារាស្ត្រពិត តែគាត់គិតកសាងកុសលនេះល្អប្រសើរណាស់» (S18¶4)។ ពាក្យនេះទាំងបញ្ជាក់ឋានៈទាប ទាំងសរសើរ។
- ¶5 «រក្សាអាណាប្រជានុរាស្ត្រ ជាសុខក្សេមក្សាន្ត»។

### សំឡេងអ្នករៀបរៀង និងប្រភពដែលដកស្រង់ (ពង្សាវតារ សាស្ត្រា ការពន្យល់ពាក្យ ការនិយាយទៅអ្នកអាន)
- ¶5 វង់ក្រចក «(ព្រះបាទពញាយ៉ាត)» ជាការពន្យល់របស់អ្នករៀបរៀង។
- ¶6 ដកស្រង់ **សាស្ត្រាវត្តកំពង់ត្រឡាចក្រោម**៖ រាជពង្សាវតារ (ព្រះនាមស្ដេច និងសករាជ) ផ្អែកលើ «អក្សរថ្ម» ដែលស្ដេចព្រះសុគន្ធាធិបតី ព្រះនាមប៉ាន ចម្លងពីមហានគរ។ សករាជក្នុងអក្សរថ្ម «ឃើញលឿនលើសទៅ» ព្រោះ «ពុំបានកាត់បច្ចុប្បន្ន»។ សាស្ត្រានោះសុំ «អ្នកប្រាជ្ញទាំងឡាយពិចារណា»។ នេះជាភស្តុតាងសំខាន់ថា សាស្ត្រាដើមខ្លួនឯងដឹងពីបញ្ហាកាលប្បវត្តិ ហើយអះអាងថាប្រើសិលាចារឹកជាប្រភព។
- ¶3 សម្ដីរបស់មន្ត្រីដាក់ក្នុងវង់ក្រចក ហើយ ¶4 គំនិតរបស់ព្រះរាជាក៏ដាក់ក្នុងវង់ក្រចកដែរ។

- **ស្ត្រី៖** ដូនពេញ «ឃោសនាអ្នកស្រុក» (¶3) ហើយព្រះរាជាដាក់ឈ្មោះវត្ត និងខេត្តតាមនាមគាត់ (¶4)។ ព្រះមាតា សម្ដេចព្រះមាលាទេវី សាងវាំងផ្ទាល់ខ្លួន (¶1)។ ព្រះមហេសី និងស្នំ ចូលរួមជញ្ជូនដី (¶4)។
- **ប្រជាជន៖** រាស្ត្រចូលរួមដោយស្ម័គ្រចិត្ត បន្ទាប់ពីព្រះរាជាធ្វើជាគំរូ (¶4)។ «ការបកស្រាយ»៖ នេះជាឧទាហរណ៍កម្រនៃការដឹកនាំដោយគំរូ ដែលប្រជាជនសាមញ្ញជាអ្នកផ្តួចផ្តើម។
- **បរទេស៖** «រាជសត្រូវ» មិនបញ្ជាក់ឈ្មោះ (¶4)។

## S19 — ផែនដីព្រះសេរីរាជា (សាស្ដ្រាខ្លះថា ព្រះរាជាអង្គនេះព្រះនាមស្រីរាជា) (គ.ស.១៤៣៣)
**ស្ថានភាពអាន៖** បានអានគ្រប់កថាខណ្ឌ ¶1–¶4

### សង្ខេប
ព្រះស្រីរាជាឡើងរាជ្យនៅព្រះជន្ម ៣០ ស្នងព្រះជេដ្ឋា។ ទ្រង់សាងព្រះមេរុ ហើយធ្វើបុណ្យព្រះបរមសពព្រះជេដ្ឋា ៧ ថ្ងៃ។ ក្រោយមកទ្រង់សោយរាជ្យជាសុខ (¶1)។ ២០ ឆ្នាំក្រោយមក ទ្រង់ដឹងថាស្រុកសៀមមាន «ចលាចល»។ អ្នករៀបរៀងពន្យល់ថា ក្នុងបរិបទនេះ ពាក្យនោះមានន័យថាសង្គ្រាមក្នុងស្រុក។ ទ្រង់ប្រជុំព្រះរាជវង្ស និងមន្ត្រី (¶2)។ ទ្រង់ត្រាស់រំលឹកថា សៀមវាយកម្ពុជាពីរដងហើយ ហើយកៀរគ្រួយកទៅច្រើន រហូតព្រះបិតាត្រូវថយមកចតុម្មុខ។ ឥឡូវ សៀមមានវិបត្តិ ព្រោះឃុនវរវង្សក្បត់ សម្លាប់ព្រះចៅយ៉តហ្វា ហើយឡើងរាជ្យ។ ទ្រង់ស្នើឲ្យ «លើកទៅសងសឹកវិញ» ព្រោះជឿថានឹងឈ្នះ (¶3)។ មុនពេលព្រះរាជវង្ស ឬសេនាបតីឆ្លើយ ពួកសេនាទាហានទូលភ្លាមថា ពេលមុនដែលនគរបែក គឺមកពីពួកខ្លួននៅឆ្ងាយ មកជួយមិនទាន់។ ឥឡូវពួកខ្លួនមានគ្រប់គ្រាន់ ហើយត្រៀមទទួលអាសា (¶4)។ ផ្នែកនេះបញ្ចប់ត្រង់នោះ ហើយ S20 បន្តរឿង។

## S20 — ផែនដីព្រះសេរីរាជា (តពីលេខ៤) (សាស្ដ្រាខ្លះថា ព្រះរាជាអង្គនេះព្រះនាមស្រីរាជា) (គ.ស.១៤៣៣)
**ស្ថានភាពអាន៖** បានអានគ្រប់កថាខណ្ឌ ¶1–¶39

### សង្ខេប
**ការវាយសៀម៖** ព្រះស្រីរាជាកេណ្ឌពល ១២ ម៉ឺននាក់។ ទ្រង់ទុកព្រះធម្មរាជាឲ្យរក្សាព្រះនគរ ហើយចាត់មេទ័ពតាមទិស (¶1)។ ទ័ពវាយតាមផ្លូវពោធិ៍សាត់-បាត់ដំបង ដល់មឿងបស្ចឹម ហើយកៀរគ្រួសៀមបាន «ជាងមួយពាន់ប្លាយ» (¶2)។ ឈ្លើយសៀមទូលថា វិបត្តិក្នុងស្រុកសៀមបានបញ្ចប់ហើយ (¶3)។ ព្រះរាជាមិនជឿ ទើបចាត់អ្នកស៊ើបការណ៍ ហើយកៀរគ្រួបានបួនពាន់ប្លាយទៀត (¶4)។ ព្រះរាជពង្សាវតារភាគ ៣ ផ្តល់ព័ត៌មានបន្ថែមពីការស៊ើបការណ៍ (¶5)។ ចៅហ្វាទឡ្ហៈវាយយករយង និងចន្ទបូរី រួចសុំទុកទ័ពរក្សាទីនោះ ហើយព្រះរាជាយល់ព្រមដោយមានលក្ខខណ្ឌ (¶6–¶8)។ ទ័ពខ្មែរគ្មានកាំភ្លើងធំ ដូច្នេះចូលជិតក្រុងសៀមមិនបាន។ ចក្រីប៉ែនស្លាប់ (¶9)។ ព្រះរាជាប្រឹក្សា ហើយថយទ័ព (¶10)។ ស្ដេចហង្សាវតីក៏ថយវិញដែរ ដោយខ្លាចទ័ពខ្មែរ (¶11)។

**ការបះបោរនៅផ្ទះ៖** មានបុព្វនិមិត្ត (¶12)។ ព្រះស្រីសុរិយោទ័យ «កើតព្រះទ័យលោភ» ហើយបះបោរ ដណ្តើមយកខេត្តត្រើយខាងកើត និងខាងជើង (¶13)។ ព្រះធម្មរាជាចាត់មន្ត្រីទៅរក្សាខេត្តដែលនៅសល់ (¶14)។ ព្រះស្រីរាជាសុបិនឃើញព្រះជេដ្ឋាយកព្រះខាន់មកកាប់ព្រះអង្គ ហើយហោរាទាយខុសគ្នា។ ដំណឹងពីការបះបោរមកដល់ (¶15)។ ទ្រង់ត្រឡប់មកវិញ (¶16)។ ព្រះធម្មរាជាផ្ញើសំបុត្រ ហើយព្រះស្រីរាជាសន្យាតាំងព្រះអនុជជាមហាឧបរាជ (¶17–¶19)។ ទ្រង់ឈ្នះព្រះស្រីសុរិយោទ័យម្តង (¶20–¶21) តែបរាជ័យជាច្រើនដងក្រោយមក (¶23)។ ទាហានរត់ចោល ព្រោះព្រះស្រីសុរិយោទ័យដាក់ទោសគ្រួសាររបស់ពួកគេ ដូច្នេះទ្រង់ថយទ័ព (¶24)។

**ព្រះធម្មរាជាដាច់ចេញ៖** ព្រះធម្មរាជាសម្រេចពឹងខ្លួនឯង (¶25–¶26)។ ទ្រង់បញ្ជូនទ្រព្យ និងស្នំទៅថ្វាយព្រះស្រីរាជា តែទុកព្រះខាន់ជ័យមង្គល និងគ្រឿងរាជ្យនៅនឹងព្រះអង្គ (¶27–¶28)។ សំបុត្ររបស់ទ្រង់សុំកុំឲ្យព្រះស្រីរាជាបញ្ជូនមន្ត្រីមក (¶29–¶34)។ ព្រះស្រីរាជាខ្ញាល់ តែមន្ត្រីទូលទប់ ព្រោះខ្លាចមានសត្រូវពីរខាង។ ទ្រង់យល់ព្រម ហើយចែកខេត្ត (¶35–¶37)។ ទ័ពនៅបាសានអញ្ជើញព្រះស្រីសុរិយោទ័យឡើងរាជ្យ ព្រោះគេជឿថាជាវង្សបងច្បង ហើយសង្ឃឹមបានឋានៈ (¶38–¶39)។

### ការប្រឹក្សា ទំនាយ សាសនា ពិធី
- ¶2 «លុះបានព្រះពិជ័យឫទ្ធិហើយ» (ពិធីមុនចេញទ័ព)
- ¶10 ការប្រឹក្សាជាមួយមេទ័ព ហើយសម្រេចថយ
- ¶12 បុព្វនិមិត្ត៖ «ព្រះខាន់ឡើងច្រែះ ព្រះជ្រៃរេចស្លឹក» (S20¶12)
- ¶15 សុបិន៖ ព្រះជេដ្ឋាដែលសុគតយកព្រះខាន់មកកាប់ព្រះអង្គជា ៣ កំណាត់។ ហោរាទាយពីរបែប (ឈ្នះមារសត្រូវ / វិវាទជាមួយព្រះរាជវង្ស)។ ការទាយទីពីរត្រូវនឹងដំណឹងដែលមកដល់ ៥ ថ្ងៃក្រោយ។
- ¶22 បុព្វនិមិត្តផ្កាយដុះកន្ទុយនៅទិសបូព៌ ដែលកន្ទុយចាក់ទៅទិសទក្សិណ ១៥ ព្យាម
- ¶25–¶26 ការប្រឹក្សារបស់ព្រះធម្មរាជា
- ¶28 «ព្រះជ័យមង្គល និង គ្រឿងបញ្ចក្សត្រ» ជាវត្ថុស្របច្បាប់នៃរាជ្យ។ «ការបកស្រាយ»៖ ការទុកវត្ថុទាំងនេះ គឺជាការប្រកាសភាពស្របច្បាប់ដោយស្ងៀមស្ងាត់ (ភ្ជាប់ទៅ S22)។
- ¶36–¶37 មន្ត្រីទូលទប់ ហើយព្រះរាជាទទួលយក
- ¶38 ការជំនុំរបស់ទ័ព (ការអញ្ជើញឡើងរាជ្យដោយក្រុមយោធា)

### តួលេខ (កងទ័ព ដំរី ឆ្នាំ អាយុ)
- ទ័ពខ្មែរ ១២ ម៉ឺននាក់ (១២០,០០០) ព្រមទាំងរទេះគោ សេះ និងដំរី (¶1, ¶11)
- គ្រួសៀមដែលកៀរបាន «ជាងមួយពាន់ប្លាយនាក់» (¶2) + «ជាងបួនពាន់ប្លាយនាក់» (¶4)
- ចក្រីប៉ែនជាមួយពល ៥០០ (¶9); ថយទៅចម្ងាយ «២០០» (មិនបញ្ជាក់ឯកតា) (¶9)
- ទ័ពហង្សាវតី ៣ ម៉ឺន (¶11)
- ពលបាត់ដំបង ៥០០០ (¶16)
- កន្ទុយផ្កាយ ១៥ ព្យាម (¶22)
- អ្នកឃុំសម្លាប់ ៤ នាក់ (¶5)

### ភាសាវាយតម្លៃ (សរសើរ/មើលងាយ ឧ. «អា…»)
- ព្រះស្រីសុរិយោទ័យ៖ «ស្ដេចកើតព្រះទ័យលោភចេតនា… គិតក្បត់ដណ្ដើមរាជ្យ» (S20¶13); «ទ្រង់កើតព្រះទ័យក្បត់» (¶15 សម្ដីអ្នកនាំសារ); «ចេញក្បត់» (¶17 សំបុត្រព្រះធម្មរាជា)។ គួរកត់សម្គាល់ថា អត្ថបទនៅតែប្រើរាជស័ព្ទពេញលេញចំពោះព្រះអង្គ។
- សៀម៖ «ខាងសៀមវិញវាមានកាំភ្លើងធំច្រើន វាបាញ់…» (¶9) ជាសំឡេងអ្នកនិទាន។
- សម័យរបស់ព្រះស្រីរាជាត្រូវហៅថា «កលិយុគ» (¶15)។
- ទាហានរបស់ព្រះស្រីសុរិយោទ័យ៖ ចិត្ត «ស្ម័គ្រស្មាគម» តែក៏មានការលោភឋានៈ (¶38)។
- មិនឃើញ «អា» ទេ។

### សំឡេងអ្នករៀបរៀង និងប្រភពដែលដកស្រង់ (ពង្សាវតារ សាស្ត្រា ការពន្យល់ពាក្យ ការនិយាយទៅអ្នកអាន)
- ¶1 ការប្រៀបធៀបជាមួយសម័យថ្មី៖ ចៅហ្វាទឡ្ហៈ «មានបណ្ដាស័កស្មើនឹងនាយករដ្ឋមន្ត្រី ឬ រដ្ឋមន្ត្រីទី១»។
- ¶1 «មិនដឹងនាមអ្វីទេ ព្រោះសស្ត្រាទាំងអស់ឥតមានដាក់នាមលោកសោះ» (អំពីយមរាជ)។
- ¶2 «រណ + យុទ្ធ = សង្គ្រាម»; «មឿងប្រាជិមសព្វថ្ងៃនេះ»។
- ¶5 «សេចក្ដីបន្ថែម» ពី **ព្រះរាជពង្សាវតារភាគ៣ ព្រះរាជបណ្ណាល័យលេខ គ៥៣-៣** ដែល «មានសេចក្ដីលើសពីពង្សាវតារវត្តកំពង់ត្រឡាច»។ នេះបញ្ជាក់ថាតួអត្ថបទមេគឺសាស្ត្រាវត្តកំពង់ត្រឡាច។
- ¶11 «ព្រះចៅពិរោរាជា (តិលោករាជ?)» ជាការសង្ស័យរបស់អ្នករៀបរៀង។
- ¶14 «ខេត្តទឹកខ្មៅ (នៅដែនកូសាំងស៊ីនសព្វថ្ងៃនេះ)»។
- ¶20, ¶29 «ខែត្របរិបូណ៌សព្វថ្ងៃនេះ»។
- ¶24 **លេខយោងវែងអំពីកាលប្បវត្តិ**៖ សាស្ត្រាស្លឹករឹត «តាំងពីរាជ្យព្រះបាទអង្គជ័យ… មិនសូវត្រូវគ្នាសោះ» ខុសគ្នា ១០, ២០ ឬ ៣០ ឆ្នាំ។ អ្នករៀបរៀង «មិនហ៊ានកែប្រែ» ហើយទុកតាមសាស្ត្រាវត្តកំពង់ត្រឡាចក្រោម រង់ចាំ «គណៈកម្មការខាងប្រវត្តិវិជ្ជា» ហើយនិយាយទៅកាន់ «លោកអ្នកអាន»។ នេះជាសេចក្តីថ្លែងវិធីសាស្ត្រដ៏សំខាន់បំផុតរបស់អ្នករៀបរៀង សម្រាប់ផ្នែក ៨ (ការវិភាគប្រភព)។ «ការបកស្រាយ»៖ ពាក្យ «គណៈកម្មការខាងប្រវត្តិវិជ្ជា» អាចជួយកំណត់សម័យរៀបរៀង។

### ស្ត្រី ប្រជាជនសាមញ្ញ ជនជាតិ/បរទេស
- **ស្ត្រី៖** ព្រះធម្មរាជាបញ្ជូន «ព្រះរាជទេពីនារីស្រីព្រះស្នំ» ទៅព្រះស្រីរាជាដូចជាវត្ថុនយោបាយ (¶27)។ ពួកនាង «យំក្រាបទៀបព្រះបាទ… ស្រដីរអាក់រអួល» (¶29)។ នេះជាសំឡេងអារម្មណ៍កម្ររបស់ស្ត្រី។ ក្នុងសៀម នាងស្រីសុតាចន្ទ្រ៍មានតួនាទីក្នុងការក្បត់ (¶3, ¶5)។
- **ប្រជាជន៖** ការកៀរគ្រួសៀម ៥០០០+ នាក់ (¶2, ¶4) ហើយទុកនៅពោធិ៍សាត់ (¶20)។ រាស្ត្រខ្មែរនៅរយង «ចុះចូល» ចៅហ្វាទឡ្ហៈ (¶6) ដែលបង្ហាញថាមានខ្មែររស់នៅក្នុងដែនសៀម។ កេណ្ឌរាស្ត្រពោធិ៍សាត់ជាអ្នកដឹកស្បៀង (¶20)។ **ទាហានរត់ចោលទ័ពទៅរកប្រពន្ធកូន** ពេលព្រះស្រីសុរិយោទ័យដាក់ទោសគ្រួសារ (¶24)។ នេះជាភស្តុតាងនៃការសម្រេចចិត្តរបស់មនុស្សសាមញ្ញ ក្នុងការរកឃើញលើ ៨.៥។ រដូវវស្សាកំណត់សង្គ្រាម (¶24)។
- **ជនជាតិ/បរទេស៖** សៀម (អយុធ្យា) មានកាំភ្លើងធំ (¶9)។ ភូមា (ហង្សាវតី) (¶11)។ កូសាំងស៊ីន (¶14)។

### កំហុសអត្ថបទ និងភាពមិនស៊ីគ្នា
- ¶3 «នាងជាយា ស្រីសុតាចន្ទ្រ៍» ទល់នឹង ¶5 «មាតាស្រីសុតាចន្ទ្រ៍»។ ¶3 «ឃុនពិរេន្ត្រទេព» ទល់នឹង ¶5 «ឃុនពិរេន្ទ្រទេព»។ ¶3 «ព្រះចៅចក្រពត្តិ» ទល់នឹង ¶5 «ព្រះឌៀររាជា… ព្រះមហាចក្រពត្រាធិរាជ»។
- ¶1 ចៅហ្វាទឡ្ហៈជាមេកងទ័ពមុខ (ផ្លូវគោក) តែ ¶6 ហៅលោកថា «មេទ័ពជើងទឹក» នៅរយង។
- ¶2 «ពោធិ៍សាត់ បាត់តំបង» (សរសេរ «តំបង») ទល់នឹង ¶16 «បាត់ដំបង»។ «មឿងបស្ចឹម / បស្ចិម / បច្ឆិមបុរី» មានអក្ខរាវិរុទ្ធខុសៗគ្នា។
- ¶9 «បោះចម្ងាយ២០០សិន» គ្មានឯកតា។
- ¶19 ការសន្យាតាំងព្រះធម្មរាជាជាមហាឧបរាជ មិនឃើញគេអនុវត្តទេ។
- ¶21 ឆ្នាំ «កុរបញ្ចស័ក» សម្រាប់ការប្រយុទ្ធ មុន ¶24 «ឆ្នាំរកា សប្តស័ក» ១៤៦៥។ តាមលំដាប់ស័ក ប្រហែលជាត្រូវ តែគម្លាតពី ១៤៥៣ ដល់ ១៤៦៥ (១២ ឆ្នាំ) មិនច្បាស់ទេ ដោយសារអត្ថបទរៀបរាប់ហាក់ដូចជាព្រឹត្តិការណ៍ជាប់ៗគ្នា។
- ¶24 ព.ស. ២០០៨ មិនស៊ីនឹង គ.ស. ១៤៦៥ ទេ (គួរជា ២០០៩)។
- ¶37 ព្រះស្រីរាជាគ្រប់គ្រងខេត្តលង្វែក ពោធិ៍សាត់ បាត់ដំបង សៀមរាប… (ទិសលិច/ជើង) តែ ¶37 ចុងក្រោយថា «ខេត្តក្រៅពីនោះ ត្រើយខាងលិច បានទៅព្រះធម្មរាជា»។ ឯ S22¶1 ហៅព្រះស្រីរាជាថា «អម្ចាស់ផែនដីឧត្តរទិស»។ ពាក្យសម្គាល់ទិសមិនជាប់លាប់។
- ¶3 មានសញ្ញាសម្រង់ពីរដង « " " »។ ¶6, ¶25, ¶31, ¶32 មាន «\» ដែលជាសំណល់ទម្រង់អត្ថបទ។
- ព្រះធម្មរាជាត្រូវបានទុកឲ្យរក្សានគរនៅ ១៤៥៣ (¶1) តែ S23¶1 ថាព្រះអង្គមានព្រះជន្ម ២៦ នៅ ១៤៧៨ (ពោលគឺកើតប្រហែល ១៤៥២)។ ដូច្នេះមិនស៊ីគ្នា (មើល S23)។

### ការភ្ជាប់ទៅករណី (C/K) និងករណីថ្មីដែលស្នើ
- **C06**៖ ផ្នែកនេះជាស្នូលនៃ C06។ វាមានការសម្រេចចិត្តច្រើន ហេតុផលច្បាស់ ការប្រឹក្សាច្រើនដង (¶10, ¶25–¶26, ¶36) ទំនាយ (¶12, ¶15, ¶22) ការសរសេរសំបុត្រ (¶17–¶18, ¶30–¶34) និងផលវិបាក។ ត្រូវកែ C06៖ ជម្លោះគឺជាការប្រកួតប្រជែងបីភាគី (ព្រះស្រីរាជា / ព្រះស្រីសុរិយោទ័យ / ព្រះធម្មរាជា) មិនមែនគ្រាន់តែ «ការចាប់ព្រះស្រីរាជា» ទេ។ ការចាប់នោះកើតឡើងនៅ S23។
- **K11**៖ ការនាំព្រះស្រីរាជា និងពញាអុង (ពញាឱង) ទៅអយុធ្យា **មិនមានក្នុង S19–S20 ទេ**។ វានៅក្នុង S23¶31–¶40។ ត្រូវកែទីតាំងរបស់ K11។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-៦៖** «ព្រះធម្មរាជារក្សាគ្រឿងរាជ្យ និងសំបុត្រ "សុំកុំឲ្យបណ្ដាជាខ្ញុំទៅមក"» (¶27–¶34) ជាការដាច់ចេញដោយសន្តិវិធី។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-៧៖** «មន្ត្រីទូលទប់ព្រះស្រីរាជាកុំឲ្យបើកសមរភូមិទីពីរ» (¶35–¶37) ជាការប្រឹក្សាដែលកែប្រែការសម្រេចចិត្តរបស់ព្រះរាជា (ជំពូក ៥)។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-៨៖** «ការដាក់ទោសគ្រួសារ និងការរត់ចោលទ័ព» (¶24) តម្លៃមនុស្ស និងភាពស្មោះត្រង់ (ភ្ជាប់ទៅ K24)។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-៩៖** «ការអញ្ជើញឡើងរាជ្យដោយទ័ព ផ្អែកលើសិទ្ធិវង្សបងច្បង និងការសង្ឃឹមបានឋានៈ» (¶38–¶39)។
- ¶9 កង្វះកាំភ្លើងធំ ជាចំណុចសម្រាប់ជំពូក ៦ (សមត្ថភាពយោធា)។

### សង្ខេប
ព្រះស្រីសុរិយោទ័យ ជាព្រះរាជបុត្រព្រះនារាយណ៍រាជា ឡើងរាជ្យនៅព្រះជន្ម ២៦ ក្នុងឆ្នាំរកា នពស័ក ព.ស. ២០២១ គ.ស. ១៤៧៧។ ទ្រង់តែងតាំងចៅហ្វាយស្រុកឲ្យរក្សាខេត្តពីសៀមបូក ក្រចេះ បាសាន ព្រៃវែង បាភ្នំ… រហូតដល់ព្រំដែនចាម (¶1)។ មន្ត្រីនៅចតុម្មុខឃើញថាក្សត្រពីរអង្គទៀតសុទ្ធតែទ្រង់រាជ្យ ហើយមានកិត្តិយសខ្ពស់ជាងម្ចាស់របស់ខ្លួន ទើបអញ្ជើញព្រះធម្មរាជាឲ្យឡើងរាជ្យ។ ព្រះធម្មរាជាបដិសេធ ហើយគ្រាន់តែរក្សាខេត្តពីសំរោងទង បាទី… កំពត កំពង់សោម រហូតដល់សមុទ្រ (¶2)។

បន្ទាប់មក អ្នករៀបរៀងប្តូរទៅរឿងយួន និងចាម។ ចាមចាញ់យួន ហើយយួនតាំងកូនស្ដេចចាមជាស្ដេចចំណុះ។ ស្ដេចចាមម្នាក់ (ចៅរាំងឡាក់ក្រាទ្រុត ឬប៉ោធុន) លើកពល «១០ សែន» វាយយកបានពីរខេត្ត។ ស្ដេចយួន ឡេ ចាញ់តុង វាយបីផ្លូវ ហើយឈ្នះ។ នៅ គ.ស. ១៤៧២ ស្ដេចចាមផ្ញើសុភអក្សរតាមរយៈគយបារៀ-ដូនណៃ មកសុំទ័ពពីកម្ពុជា។ ព្រះសុរិយោទ័យបដិសេធ ព្រោះ «ស្រុកយើងកំពុងមានទ័ពច្បាំងរាំងជល់ មិនទាន់រាបទាប»។ ទ្រង់ឲ្យបម្រើស្ដេចចាមត្រឡប់ទៅវិញ។ ក្រោយមក យួនវាយបែកនគរចាម ចាប់ស្ដេចចាម ហើយចែកដែនជាខេត្ត។ រាស្ត្រ និងព្រះវង្សចាមជាច្រើនរត់មកស្រុកខ្មែរ ហើយខ្លះទៀតទៅស្ទៀង រដែ (¶3)។

### សង្ខេប
ព្រះស្រីរាជា ដែលអត្ថបទហៅថា «អម្ចាស់ផែនដីឧត្តរទិស» ផ្ញើសុភអក្សរនៅឆ្នាំជូត សំរឹទ្ធិស័ក ច.ស. ៨៣០។ ក្នុងសំបុត្រនោះ ទ្រង់សុំព្រះខាន់រាជជ័យមង្គល និងដាវ និងក្រឹសដែកសម្រាប់ពលសេះ និងដំរី ដែលនៅក្នុងព្រះឃ្លាំង (¶1)។ ព្រះធម្មរាជាត្រាស់ថា របស់ទាំងនេះ «ជារបស់សម្រាប់ព្រះមហានគរ» ហើយមិនគួរថ្វាយទេ។ ទ្រង់ផ្ញើតែអាវក្រឹសដែក ស្រកាពង្រូលសម្រាប់ពលរទេះ ២០ សម្រាប់ និងមួកដែក ២ ជំនួសវិញ (¶2)។ ព្រះស្រីរាជាខ្ញាល់ណាស់ ព្រោះព្រះខាន់មិនបានមកវិញ ហើយអាវក្រឹសសម្រាប់ពលរទេះនោះ «គេលែងប្រើអស់ទៅហើយ»។ ទ្រង់បញ្ជូនបម្រើទៅសុំច្រើនដងទៀត តែសុំមិនបាន (¶3)។ មន្ត្រីនៅចតុម្មុខ «ដោយស្រឡាញ់អម្ចាស់របស់ខ្លួនណាស់» អញ្ជើញព្រះធម្មរាជាឲ្យឡើងរាជ្យ ហើយព្រះអង្គបដិសេធម្តងទៀត។ នៅឆ្នាំរកា នពស័ក ព្រះសុរិយោទ័យឡើងរាជ្យ ហើយមួយឆ្នាំក្រោយមក «ព្រះសេរីធម្មរាជា» ទើបព្រមទទួលរាជ្យ (¶4)។

## S23 — ផែនដីព្រះធម្មរាជា (ពីគ.ស.១៤៧៨ ដល់ គ.ស.១៥០៤)
**ស្ថានភាពអាន៖** បានអានគ្រប់កថាខណ្ឌ ¶1–¶41

### សង្ខេប
ព្រះធម្មរាជាឡើងរាជ្យនៅ គ.ស. ១៤៧៨ ព្រះជន្ម ២៦។ ទ្រង់តាំងអគ្គមហេសី ហើយប្រទានយសដល់ព្រះវង្ស។ នៅឆ្នាំនោះមានរញ្ជួយដីជាង ៥ ម៉ោង (¶1)។ ព្រះស្រីរាជាចាត់ទុកការឡើងរាជ្យនេះជាការក្បត់ ហើយចង់វាយភ្នំពេញ តែត្រូវងាកទៅទប់ព្រះស្រីសុរិយោទ័យវិញ។ សង្គ្រាមមិនមានអ្នកឈ្នះចាញ់ ហើយភ្លៀងបង្ខំឲ្យថយ (¶2)។ ផែនដីបែកជាបី ហើយសង្គ្រាម ១០ ឆ្នាំបណ្តាលឲ្យរាស្ត្រស្លាប់ច្រើន (¶3, ¶6)។ ពញាឱងរាជាប្រសូត ហើយមានការវាយប្រហារម្តងទៀត (¶4)។ ព្រះធម្មរាជាធ្វើតែការពារ (¶5, ¶8)។ ដោយអាណិតរាស្ត្រ ទ្រង់ផ្ញើព្រះរាជសារសុំជំនួយអយុធ្យានៅ ១៤៨៥ (¶6, ¶9–¶15)។ ស្ដេចសៀមសួរថាតើព្រះអង្គមានព្រះមាតាជាសៀមឬទេ ហើយអ្នករៀបរៀងរំលឹកអ្នកអានពីនាងស៊ីសាងាម (¶16–¶18)។ ទ័ពជើងទឹក និងជើងគោកសៀមចូលមក (¶19–¶21)។ ព្រះធម្មរាជាជួបស្ដេចសៀម ហើយហៅគ្នាថា «រៀម/អនុជ» (¶22–¶23)។ ពួកគេឡោមព្រះស្រីរាជា (¶24)។ ចៅហ្វាទឡ្ហៈទូលថាតទល់មិនបាន ឯទាហានចង់ច្បាំង (¶25–¶27)។ ព្រះស្រីរាជាសម្រេចចុះចូល «ព្រោះអាណិតអស់រាស្ត្រប្រជា» (¶28–¶29)។ ក្សត្របីអង្គជួបគ្នាដោយគ្មានអាវុធនៅរោងទង (¶30–¶31)។ សៀមឃុំព្រះស្រីរាជា (¶32)។ ព្រះស្រីសុរិយោទ័យរីករាយ ហើយមកជួប ដោយសង្ឃឹមបានរាជ្យ តែត្រូវឃុំដែរ (¶33–¶35)។ ក្សត្រទាំងពីរត្រូវនាំទៅ (¶36)។ ក្សត្រប្តូររាជបណ្ណាការគ្នា ហើយគ្រួសៀមត្រូវប្រគល់ឲ្យវិញ (¶37–¶39)។ ព្រះស្រីរាជាសុគតតាមផ្លូវ ហើយពញាឱងទៅជាព្រះរាជបុត្រធម៌របស់ស្ដេចសៀម។ ព្រះស្រីសុរិយោទ័យសុគតនៅអយុធ្យា (¶40)។ ព្រះធម្មរាជាលើកលែងពន្ធ ៣ ឆ្នាំ (¶41)។

- ¶21 ថ្ងៃអាទិត្យ ៦ កើត ខែបុស្ស ឆ្នាំម្សាញ់ បញ្ចស័ក ព.ស. ២០២៩ គ.ស. ១៤៨៥ ម.ស. ១៤០៧ ច.ស. ៨៤៧ «វេលាព្រឹកម៉ោង ២ (ម៉ោងបុរាណ ត្រូវម៉ោង ៩ កន្លះ…)»
- ¶30 ថ្ងៃ ៣ កើត ខែមាឃ ឆ្នាំម្សាញ់ សប្តស័ក; ¶31 ថ្ងៃ ១០ រោច ខែមាឃ
- ¶40 «បញ្ឈប់ទ័ពនៅទីនោះ ៣ ថ្ងៃ»; «បញ្ឈប់ទ័ពបាន ១៥ ថ្ងៃ»
- ¶41 លើកលែងពន្ធ ៣ ឆ្នាំ

### ការប្រឹក្សា ទំនាយ សាសនា ពិធី
- ¶1 «អព្ភូតហេតុ» រញ្ជួយដី «ជាង ៥ ម៉ោង» នៅឆ្នាំឡើងរាជ្យ (អត្ថបទមិនបកស្រាយន័យទេ)។
- ¶13–¶14 ព្រះរាជសារប្រើហេតុផលសាសនា៖ «សមណជីព្រាហ្មណ៍ ទាំងព្រះពុទ្ធសាសនាក៏សៅហ្មង» ហើយសុំឲ្យស្ដេចសៀម «ជួយទំនុកបំរុង ព្រះពុទ្ធសាសនា»។
- ¶14 គោលការណ៍ «ព្រះនគរទាំងពីរ ទុកជាផែនដីមួយ» ជួយគ្នាដូចបងប្អូន «តាមព្រះរាជប្រវេណី»។
- ¶22 ពិធីចេញទ័ព៖ «ពិជ័យឫក្សល្អ» ហោរាគោះគង ហើយមានស្វេតច្ឆ័ត្រ។
- ¶25–¶29 ការប្រឹក្សារបស់ព្រះស្រីរាជា៖ យោបល់ពីរខាង (ចៅហ្វាទឡ្ហៈ ទល់នឹងទាហាន) ហើយព្រះរាជាថ្លឹងថ្លែង (¶28)។ នេះជាការប្រឹក្សាដែលមានការជជែកវែកញែកច្បាស់ជាងគេក្នុង G2។
- ¶31 ការជួបជុំ «ពុំកាន់គ្រឿងអាវុធឡើយ» ក្នុងព្រះពន្លាជ័យ។
- ¶40 ព្រះមេរុ និងការរើសព្រះធាតុ (ពិធីរបស់ព្រះស្រីសុរិយោទ័យនៅអយុធ្យា)។
- ¶41 សាលាឆទាន និងទសពិធរាជធម៌។

### តួលេខ (កងទ័ព ដំរី ឆ្នាំ អាយុ)
- ព្រះជន្ម L025 ២៦ ពេលឡើងរាជ្យ (¶1)
- រញ្ជួយដីជាង ៥ ម៉ោង (¶1)
- សង្គ្រាម ១០ ឆ្នាំ (¶3, ¶6); រាជ្យ L023 ២០ + ១០ = ៣០ ឆ្នាំ (¶6); L024 ៥ ឆ្នាំ (¶7); L025 ៨ ឆ្នាំ (¶8)
- រាជបណ្ណាការទៅអយុធ្យា៖ ចានមាស ៤ ចានប្រាក់ ៤ កុយរមាស ៤ ភ្លុកដំរីធំ ៤ សំពត់ហូល ៤០ សូត្រ ៤ ហាប (¶15)
- ទ័ពជើងទឹកសៀម ២ ម៉ឺន (២០,០០០) (¶19) ទល់នឹង ៣០០០ តាមភាគ ៣ (¶20)
- ដំរីព្រះទីនាំង កំពស់ «ប្រាំហត្ថមួយចំអាម» (¶21)
- អ្នកនាំសារ ២ នាក់ (¶29); ក្សត្រ ៣ អង្គ (¶30–¶31)
- រាជបណ្ណាការរបស់ L025 (¶37)៖ មកុដមាស ១, សង្វារមាស ៣, ប្រាក់ ៣ ហាប, ដំរីមានភ្លុក ៦, សេះ ២០០, សូត្រ ១០ ហាប, ក្របាស់ ៥ ហាប, សំពត់ហូល ៣០០, ព្រែកសូត្រ ១០០, រង្គ ៣ ហាប, ថ្មកែវ ២០០ ដុំ ព្រមទាំងគ្រួសៀមទាំងអស់
- រាជបណ្ណាការតបពីសៀម (¶38)៖ មកុដមាស ១, វាំងណយ ៥, គ្រឿងដំរីសឹក ២, ស្គរ ១ គូ, ពិណពាទ្យ ២ វង់, ដំរីភ្លុក ១០០ ...
- ស្ដេចត្រាញ់ ៥ (មានបណ្ដាសក្ដិ ១០ ហួពាន់) (¶39)
- ពន្ធលើកលែង ៣ ឆ្នាំ (¶41)

### ភាសាវាយតម្លៃ (សរសើរ/មើលងាយ ឧ. «អា…»)
- ព្រះស្រីរាជាវាយតម្លៃព្រះធម្មរាជាថា «ប្រែព្រះទ័យក្បត់ឡើងសោយរាជ្យ» (¶2) ជាការវាយតម្លៃរបស់តួអង្គ មិនមែនរបស់អ្នកនិទានទេ។
- ទាហានខ្មែរនិយាយពីសៀម៖ «ពួកសៀមនោះ វាធ្លាប់តែចាញ់ថ្វីដៃយើងទេ» (S23¶27)។
- ព្រះស្រីរាជាវាយតម្លៃទាហាន៖ «វិស័យជាជាតិទាហាន ចិត្តមនោមានះ មិនដឹងកំលាំងគេកំលាំងឯង» (S23¶28)។
- ព្រះស្រីសុរិយោទ័យត្រូវគេពណ៌នាថា «មានព្រះអំណរណាស់» ពេលព្រះបិតុលាត្រូវគេចាប់ (¶33)។ «ការបកស្រាយ»៖ ពាក្យនេះបង្ហាញការវាយតម្លៃអវិជ្ជមានដោយប្រយោល។
- ព្រះធម្មរាជាត្រូវសរសើរ៖ «ព្រះមហាករុណា អាណិតអាណាប្រជានុរាស្ត្រពន់ប្រមាណ» (¶6); «ប្រកាន់ទសពិធរាជធម៌» (¶41)។
- ព្រះស្រីរាជា «តូចព្រះទ័យណាស់» មិនសោយ ហើយសុគត (¶40) ជាការពណ៌នាដោយក្តីអាណិត។
- មិនឃើញ «អា» ទេ។

### សំឡេងអ្នករៀបរៀង និងប្រភពដែលដកស្រង់ (ពង្សាវតារ សាស្ត្រា ការពន្យល់ពាក្យ ការនិយាយទៅអ្នកអាន)
- ¶6 ការគណនារាជ្យ «នោះបើគិតរាជសម្បត្តិ ព្រះមហាក្សត្រទាំង ៣ ព្រះអង្គ…» ជាសំឡេងអ្នកគណនា។
- ¶6 «ខែត្រគោកសេះ (ខែត្របារាយណ៍)»; ¶7 «ក្រុងបាសាន (គឺស្រុកស្រីសន្ធរ… សព្វថ្ងៃនេះ)»; ¶7 «ព្រះបិតុលា (មាខាងឪពុក)»។
- ¶9 «ឲ្យលាម (បកប្រែ)»; ¶11 «មហាតលិក (សៀមអានថា មហាតិឡុក)»។
- ¶17 **ការនិយាយទៅអ្នកអាន**៖ «លោកអ្នកអានប្រហែលជានៅចាំបានថា… ស៊ីសាងាមជាតិសៀមសុទ្ធ» (S23¶17) (សម្រង់ដែលបានរៀបរាប់ក្នុងផ្នែក ៨.១)។
- ¶20 **ព្រះរាជពង្សាវតារភាគ៣ លេខ គ ៥៣-៣** «មានសេចក្ដីដូច្នេះថែមទៀត» ដែលផ្តល់ហេតុផលសងសឹក និងតួលេខ ៣០០០។
- ¶21 ការបំប្លែងម៉ោងបុរាណ («ម៉ោង ២ … ម៉ោង ៩ កន្លះ»)។
- ¶22 «លើកទៅប្រសប់ (គឺលើកទៅជួប)»។
- ¶23 «តទៅនេះសេចក្ដីក្នុងសាស្ត្រាវត្តកំពង់ត្រឡាចក្រោម» ប្តូរប្រភពត្រលប់មកតួអត្ថបទមេ។
- ¶31 **ការសុំអភ័យទោស**៖ «សូមអភ័យទោសចុះស្រុកនេះ យើងខ្ញុំមិនស្គាល់ទេ» និង «បានស៊ើបសួរអ្នកស្រុកកំពង់ចាមដែរ» (អំពីរោងទង)។ នេះបង្ហាញថាអ្នករៀបរៀងបានធ្វើការស្រាវជ្រាវវាលផ្ទាល់។
- ¶38 «(អក្ខរាវិរុទ្ធនៃពាក្យនេះមិនឃើញមាន)», «ព្រះសែង (ដាវ)», ការពន្យល់វេយ្យាករណ៍ «ខ្មែរបូរាណយកនាម ធ្វើជាកិរិយាសព្ទ»។
- ¶39 ការពន្យល់ «ស្ដេចត្រាញ់» និងបញ្ជីឈ្មោះ ៥ នាក់ (ជាការពន្យល់ស្ថាប័នរបស់អ្នករៀបរៀង)។

### ស្ត្រី ប្រជាជនសាមញ្ញ ជនជាតិ/បរទេស
- **ស្ត្រី៖** អគ្គមហេសីមានព្រះនាម (¶1)។ ព្រះមាតាពញាឱងមានឈ្មោះ (¶4)។ **ព្រះមាតាសៀមរបស់ព្រះធម្មរាជា** ជាមូលដ្ឋាននៃសម្ព័ន្ធភាព (¶17–¶18)។ ព្រះរាជបុត្រីស្ដេចសៀមត្រូវប្រទានជាព្រះជាយាដល់ពញាឱង (¶40)។ «ការបកស្រាយ»៖ ខ្សែវង្សតាមព្រះមាតាគឺជាធនធានការទូត។
- **ប្រជាជន៖** «រេហ៍ពលរាស្ត្រប្រជា ស្លុតស្លាប់អស់ជាច្រើននាក់» (¶3)។ «អស់អាណាប្រជានុរាស្ត្រវិនាស ដោយច្បាំង និងដោយជំងឺតំកាត់» ហើយចោរប្លន់ (¶6)។ ទាហានឲ្យធ្វើស្រែក្នុងរដូវវស្សា (¶2)។ ព្រះស្រីរាជាចុះចូល «ព្រោះអាណិតអស់រាស្ត្រប្រជា» (¶29)។ **គ្រួសៀមដែលត្រូវកៀរ ត្រូវប្រគល់ទៅវិញ** (¶37)។ លើកលែងពន្ធ ៣ ឆ្នាំ (¶41)។ ផ្នែកនេះមានភស្តុតាងច្បាស់ជាងគេដែលកែតម្រូវការអះអាងក្នុង ៨.៤ ថា «សុខុមាលភាពប្រជាជនកម្របង្ហាញ»។
- **បរទេស៖** អយុធ្យាជាអ្នកសម្រុះសម្រួល ដែលក្លាយជាអ្នកចាប់ក្សត្រខ្មែរពីរអង្គ។ គ្រឿងរាជបណ្ណាការមាន «សឹងប៉ាក់មាស ទេសស្រុកជ្វា» (¶38)។ មន្ត្រីខ្មែរអាចទៅអយុធ្យាតាមស្ម័គ្រចិត្ត (¶31)។

នៅ គ.ស. ១៤៨៦ ព្រះរាជឱរស (ព្រះស្រីសុគន្ធបទ) រៀនចប់ព្រះត្រៃបិតក។ ព្រះរាជាធ្វើបុណ្យ សម្ដែងធម៌ដល់មន្ត្រីរាល់ថ្ងៃសីល ហើយប្រាថ្នាជាពុទ្ធភូមិ (¶1)។ នៅ ១៤៩៦ ព្រះមែយ្យទេពបុប្ផាប្រសូតព្រះរាជបុត្រនៅវេលាចន្ទគ្រាស ដូច្នេះហោរាដាក់ព្រះនាមថា «ពញាចន្ទរាជា»។ ជំទាវវារ ភរិយាចៅពញាយមរាជ ជាអ្នកថែរក្សា (¶2)។ ព្រះរាជាប្រជុំសង្ឃរាជ និងមន្ត្រី អំពីការសាងចេតិយតម្កល់ព្រះបរមសារីរិកធាតុ ហើយពួកគេយល់ព្រម (¶3–¶4)។ ចេតិយធំត្រូវសាងនៅអាសន្ទុក (ភូមិខ្វាវ) ដោយការកេណ្ឌរាស្ត្រ និងតុបតែងដោយមាសប្រាក់ ហើយប្រើពេលជាងពីរឆ្នាំ (¶5)។ នៅ ១៤៩៥ ឬ ១៤៩៦ មានពិធីឆ្លង។ ព្រះអដ្ឋិព្រះស្រីរាជា និងព្រះស្រីសុរិយោទ័យត្រូវនាំមកពីសៀម (តាមកំណែមួយ នៅ ១៤៩១)។ ព្រះរាជាបួស ៣ ថ្ងៃ ហើយវេរភូមិ ស្រែ ខ្ញុំ ២១ ចំនួននីមួយៗ ទុកជា «ពលព្រះស្រីរតនត្រ័យ» (¶6)។ ទ្រង់ឆ្លាក់សិលាចារឹក ហើយធ្វើបុណ្យ ៣ ខែ (¶7)។ សាស្ត្រាវត្តកំពង់ត្រឡាចក្រោមបន្ថែមរឿងពីរ៖ ការទតនគរពិជ័យ (¶8) និងរឿងព្រះអង្គរតនមាលីដែលក្រពើលេប។ ព្រះរាជាខ្ញាល់មន្ត្រីដែលធ្វេសប្រហែស ហើយ «ដាក់ទំនាយ» (¶9–¶12)។ នៅ ១៤៩៩ ឧកញ៉ាសួគ៌ាលោករាយការណ៍ពីដំរីស ហើយព្រះរាជាប្រទានរង្វាន់ដល់ហ្ម និងលើកលែងកំណែនដល់អ្នកជួយ។ ដំរីនោះស្លាប់នៅឆ្នាំច (¶13–¶16)។ នៅ ១៥០២ ទ្រង់ឲ្យជីកស្រះនៅភ្នំព្រះរាជទ្រព្យ តែប្រឈួន (¶17)។ ទ្រង់សោយរាជ្យ ២៧ ឆ្នាំ ព្រះជន្ម ៦៤ (¶18)។ ទ្រង់សុគតនៅ ១៥០៤ ហើយចៅពញាដំខត្តិយ៍ ជាបុត្រច្បង ត្រូវអញ្ជើញឲ្យឡើងរាជ្យ (¶19)។

- ¶2 ឆ្នាំរោង អដ្ឋស័ក ព.ស. ២០៤០-គ.ស. ១៤៩៦ (សោយរាជ្យ ១៩ ឆ្នាំ; ចន្ទរាជាប្រសូត)
- ¶6 ព.ស. ២០៣៩-គ.ស. ១៤៩៥-ម.ស. ១៤១៧-ច.ស. ៨៥៧ ឆ្នាំថោះ សប្តស័ក ថ្ងៃ ១១ កើត ខែអាសាធ (ព្រះរាជពង្សាវតារព្រះរាជបណ្ណាល័យ៖ ព.ស. ២០៤០-គ.ស. ១៤៩៦-ម.ស. ១៤១៨-ច.ស. ៨៥៨ ឆ្នាំរោង អដ្ឋស័ក)

- ¶13 ឆ្នាំមមែ ឯកស័ក ព.ស. ២០៤៣-គ.ស. ១៤៩៩-ម.ស. ១៤២១-៨៦១
- ¶16 ដំរីសស្លាប់ «ឆ្នាំចចត្វាស័ក»
- ¶17 ថ្ងៃ ១០ កើត ខែពិសាខ ឆ្នាំច ចត្វាស័ក ព.ស. ២០៤៦-គ.ស. ១៥០២-ម.ស. ១៤២៤-ច.ស. ៨៦៤

### ការប្រឹក្សា ទំនាយ សាសនា ពិធី
- ¶1 ការសម្ដែងធម៌ដល់មន្ត្រីរាល់ថ្ងៃឧបោសថ និងការប្រាថ្នាពុទ្ធភូមិ (ព្រះរាជាជាអ្នកបណ្តុះសីលធម៌ដល់មន្ត្រី)។
- ¶2 ការដាក់ព្រះនាមតាមចន្ទគ្រាស ដោយ «ក្រុមបារគូ បុរោហិត ហោរាចារ្យ» (K14)។
- ¶3–¶4 ការប្រឹក្សាជាមួយសង្ឃរាជ ព្រះវង្ស និងមន្ត្រី ដែលព្រះរាជាយកព្រះបាទធម្មាសោកធ្វើជាគំរូ។
- ¶5–¶7 ការសាងចេតិយ ស្តូប និងវិហារ។ ការហែព្រះសារីរិកធាតុពី «ព្រះមហានគរវត្ត»។ ការបួស ៣ ថ្ងៃ (ឆាយាត្រូវដកស្រង់)។ «ឆ្នោតព្រះរាជទាន»។ ការវេរខ្ញុំ (ទាសាទាសី)។ ការជ្រួចទឹក «ទក្សិណោទក» ហើយហៅទេវតាជាសាក្សី។ ការឆ្លាក់សិលាចារឹក។
- ¶8 ការសរសើរស្ដេចបុរាណដែលសាងដោយថ្ម «គប្បីតែស្ដេចប្រទេសរាជ ១០១ មកជាចំណុះ»។
- ¶12 ការបញ្ចុះ «ខ្មោច ព្រាយ» ក្រោមបាតសសរវិហារ និង «អថ័ន» (អ្នករៀបរៀងពន្យល់ថាជាមន្តអាគម)។ «ដាក់ទំនាយ» ជាបណ្តាសាលើមន្ត្រី។ ការវេរ ២១ ម្យ៉ាងៗ ដើម្បីឧទ្ទិសកុសលដល់ព្រះរាជបុត្រី។
- ¶15–¶16 ដំរីស «គជលក្ខណៈបរិបូណ៌» ពិធីមង្គល និងការដាក់នាម។
- ¶17 ការជីកស្រះជាទាន។ ¶18 ការសូត្រពុទ្ធគុណ ធម្មគុណ សង្ឃគុណ។
- «ការបកស្រាយ»៖ ¶16–¶17 ដំរីសស្លាប់នៅឆ្នាំច ហើយព្រះរាជាប្រឈួនក្នុងឆ្នាំច (១៥០២) ដូចគ្នា។ អត្ថបទមិនភ្ជាប់ព្រឹត្តិការណ៍ទាំងពីរនេះទេ។

### តួលេខ (កងទ័ព ដំរី ឆ្នាំ អាយុ)
- ព្រះចេតិយ ៨៤០០០ (រឿងអសោក) និង ៥០០០ ព្រះវស្សា (¶3, ¶6)
- វិហារមាន «ត្រីសូល៍កំពូលប្រាំ» ហើយស្តូបមាន ៣ ជាន់ (¶5); ការសាង «ពីរឆ្នាំប្លាយ» (¶5)
- បួស ៣ ថ្ងៃ (¶6); បុណ្យ ៣ ខែ (¶7)
- ការវេរ ២១ ម្យ៉ាងៗ (ភូមិ ស្រែ ខ្ញុំប្រុស ខ្ញុំស្រី រទេះគោ រទេះក្របី សេះ ដំរី) (¶6, ¶12); មេ ២ នាក់ (¶6)
- ក្បាលក្រពើ ៥ ហត្ថ (¶11); ព្រះពុទ្ធរូប ៥ × ៥ លោហៈ = ២៥ (¶12); សសរ «មួយរយប្លាយ» (¶12)
- ដំរីស «កំពស់បួនហត្ថប្លាយ» (¶14); រង្វាន់ «មាសមួយតឹងកូនដៃ ប្រាក់មួយតឹងកូនដៃ» (¶15)
- រាជ្យ ៨ + ១៩ = ២៧ ឆ្នាំ; ព្រះជន្ម ៦៤ (¶18); រាជ្យ ១៩ ឆ្នាំនៅ ១៤៩៦ (¶2)

### ភាសាវាយតម្លៃ (សរសើរ/មើលងាយ ឧ. «អា…»)
- សរសើរ L025 ចំពោះសទ្ធា (¶1, ¶4 «សមគួរតាមព្រះរាជឱង្ការណាស់»)។
- L025 សរសើរស្ដេចបុរាណ៖ «មានបុណ្យបារមី ពន់ប្រមាណណាស់» (S24¶8)។
- ក្រពើ «សាហាវហានក្លា» (¶11)។
- ព្រះរាជាខ្ញាល់មន្ត្រី (¶12)៖ «បើមន្ត្រីណាដើរកាត់មុខព្រះវិហារកូនអញ ឲ្យមន្ត្រីនោះអន្តរាយ»។
- មិនឃើញ «អា» ទេ។

### សំឡេងអ្នករៀបរៀង និងប្រភពដែលដកស្រង់ (ពង្សាវតារ សាស្ត្រា ការពន្យល់ពាក្យ ការនិយាយទៅអ្នកអាន)
- ¶2 «ព្រះមែយ្យ (ពាក្យសៀម… ព្រះអគ្គមហេសី ឬ ស្នំឯក)» (ដែលបានដកស្រង់ក្នុង ៨.១) និង «ជំទាវវារ (ប្រហែលជាជំទាវនោះវល្លិឬ?)»។
- ¶5 «ស្ថាបនា (ពាក្យនេះមានតាំងពីបូរាណកាលមក)» និងការពន្យល់ «សង្វា… សម្រាប់ស្ត្រីពាក់»។ «ពានព្រះស្រី (ជើងពានសម្រាប់ដាក់ស្លាបារី…)»។
- ¶6 **ការប្រៀបធៀបកំណែ**៖ ព្រះរាជពង្សាវតារព្រះរាជបណ្ណាល័យ ផ្តល់ឆ្នាំខុសគ្នា (១៤៩៥ ទល់នឹង ១៤៩៦) ហើយមានព័ត៌មានបន្ថែមអំពីព្រះអដ្ឋិ (១៤៩១)។ «ទក្សិណោទក» ត្រូវពន្យល់។
- ¶8–¶12 **«សាស្ត្រា វត្តកំពង់ត្រឡាចក្រោម មានសេចក្ដីលើសពីឯកសារ ដទៃទៀត»** ដែលសញ្ញាសម្រង់បើកនៅ ¶8 ហើយបិទនៅចុង ¶12។ ដូច្នេះ រឿងនគរពិជ័យ និងរឿងរតនមាលីទាំងមូល គឺមកពីសាស្ត្រាវត្តតែមួយ។ «នគរពិជ័យ បាអា (ប្រហែលជា វត្តនគរនៅកំពង់ចាមទេដឹង)» គឺជាការសន្និដ្ឋានរបស់អ្នករៀបរៀង។
- ¶12 «អថ័ន (មកពីពាក្យ សំស្ក្រឹត អថវ៌ន…)»។
- ¶16 «ស្រុតប្រែថាស្លាប់ ពាក្យនេះ សម្រាប់និយាយចំពោះតែដំរី»។
- ¶6 «អាស្រ័យហេតុនេះហើយ បានជាមានពលព្រះស្រីរតនត្រ័យ តាំងពីនោះរៀងមក» គឺជាការពន្យល់ប្រភពដើមស្ថាប័ន។

### ស្ត្រី ប្រជាជនសាមញ្ញ ជនជាតិ/បរទេស
- **ស្ត្រី៖** ព្រះមែយ្យទេពបុប្ផា (¶2)។ ជំទាវវារជាអ្នកថែរក្សាព្រះរាជកុមារ (¶2)។ ព្រះអគ្គមហេសីធ្វើសក្ការបូជា (¶5)។ ព្រះអង្គរតនមាលីមានឆន្ទៈផ្ទាល់ «ចង់ទៅស្រង់ទឹកទន្លេធំ ឲ្យដឹងស្គាល់ជាតិ» ហើយសុំការអនុញ្ញាត (¶9)។ ស្ត្រីបរិវារ (¶10–¶11)។ ខ្ញុំស្រី ២១ នាក់ត្រូវវេរ (¶6, ¶12)។
- **ប្រជាជន៖** កេណ្ឌរាស្ត្រសាងចេតិយ (¶5) និងជីកស្រះ (¶17)។ **ខ្ញុំ (ទាសាទាសី) ត្រូវវេរជាដង្វាយ** ហើយក្លាយជា «ពលព្រះស្រីរតនត្រ័យ» (¶6, ¶12)។ «ឆ្នោតព្រះរាជទាន បណ្ដារាស្ត្រ» (¶6)។ ហ្មទាក់ដំរីទទួលរង្វាន់ ហើយ «ពលកម្លោះ» រួចពីកំណែនរាជការ (¶15)។ «ខ្មោច ព្រាយ» ក្រោមសសរ (¶12) មិនច្បាស់ថាជាមនុស្សពិត ឬជាពិធីទេ។ ត្រូវប្រុងប្រយ័ត្នពេលបកស្រាយ។
- **ជនជាតិ/បរទេស៖** ព្រះអដ្ឋិត្រូវសុំពីស្ដេចសៀម (¶6)។ «ព្រះមែយ្យ» ជាពាក្យសៀម (¶2)។

### កំហុសអត្ថបទ និងភាពមិនស៊ីគ្នា
- ចំណងជើង «ផែនដីព្រះផែនដីព្រះធម្មរាជា» មានពាក្យ «ផែនដីព្រះ» ស្ទួន។
- **អ្នកស្នងរាជ្យ៖** ¶1 ព្រះរាជឱរសគឺ «ព្រះស្រីសុគន្ធបទ» តែ ¶19 អ្នកស្នងគឺ «ចៅពញាដំខត្តិយ៍ ជាព្រះរាជបុត្រាច្បង»។ ត្រូវពិនិត្យថាជាមនុស្សតែមួយ (L026) ឬទេ។
- ¶2 ចន្ទរាជាកើតនៅ ១៤៩៦ ហើយ «ស្ដេចបានជំទាវវារ… ទំនុកបម្រុង» ទោះបីជាទារក។ ពាក្យ «ស្ដេចបាន» ចម្លែក។
- ¶6 ឆ្នាំឆ្លង ១៤៩៥ (វត្ត) ទល់នឹង ១៤៩៦ (ព្រះរាជបណ្ណាល័យ)។ ព្រះអដ្ឋិតម្កល់នៅ ១៤៩១ មុនពិធីឆ្លង។ ¶6 «ពីព្រះបរមរាជាវាំងចតុម្មុខ មកបញ្ចុះក្នុងព្រះចេតិយបន្ទាប់» ទល់នឹង «ដំកល់នៅភ្នំពេញ» មិនច្បាស់ទីតាំង។
- ¶3 «ភ្នំសន្ទុក» ទល់នឹង ¶5 «ខេត្តអាសន្ទុក… ភូមិខ្វាវ»។
- ¶9–¶11 «ព្រះអង្គរតនមាលី» ទល់នឹង «ព្រះនតនមាលី» (¶11)។
- ¶13 «ម.ស. ១៤២១-៨៦១» ខ្វះពាក្យ «ច.ស.»។
- ¶18 «ចលាចល ៨ ឆ្នាំ + អភិសេក ១៩ ឆ្នាំ = ២៧» ស៊ីនឹង ១៤៧៨–១៥០៤ (ប្រហែល ២៦–២៧ ឆ្នាំ) តែព្រះជន្ម ៦៤ មិនស៊ីនឹងព្រះជន្ម ២៦ នៅ ១៤៧៨ (S23¶1) ទេ។
- ¶2 «សោយរាជ្យបាន ១៩ ឆ្នាំ» នៅ ១៤៩៦ (= ១៤៧៨ + ១៨/១៩) តែ ¶18 ប្រើ «១៩ ឆ្នាំ» សម្រាប់រយៈពេលអភិសេកទាំងមូលរហូតដល់ ១៥០៤។
- ¶12 សញ្ញាសម្រង់បិទ « " » នៅចុងរឿង មិនមានសញ្ញាបើកច្បាស់លាស់ក្រៅពី ¶8។

### ការភ្ជាប់ទៅករណី (C/K) និងករណីថ្មីដែលស្នើ
- **K12**៖ ¶9–¶12 (មិនមែន S23 ទេ)។ ត្រូវកែ L037 ទៅ «S24» ប៉ុណ្ណោះ។ ត្រូវកត់សម្គាល់ថារឿងនេះមកពីសាស្ត្រាវត្តកំពង់ត្រឡាចក្រោមតែមួយ (កម្រិត ឃ)។
- **K14**៖ ¶2 (ទីតាំងពិត)។
- **C08**៖ ¶1 (ព្រះស្រីសុគន្ធបទរៀនព្រះត្រៃបិតក), ¶2 (ចន្ទរាជាកើត), ¶19 (ការស្នងរាជ្យ «តាមព្រះប្រវេណី»)។ ជាផ្ទៃខាងក្រោយនៃ C08។
- **C13 / K27**៖ ¶13–¶16 ដំរីសនៅរាជ្យ L025 ជាផ្ទៃខាងក្រោយនៃតម្លៃនិមិត្តរូបនៃដំរីស (មិនមែនដំរីដែលអយុធ្យាទាមទារក្នុង S32 ទេ ព្រោះដំរីនេះស្លាប់ហើយ)។
- **C07**៖ ¶6 ការសុំព្រះអដ្ឋិពីសៀម ជាផលវិបាកវែងនៃសម្ព័ន្ធភាព។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-១៨៖** «ការសាងចេតិយអាសន្ទុក ការបួស ៣ ថ្ងៃ និងការវេរខ្ញុំជាពលព្រះស្រីរតនត្រ័យ» (¶3–¶7) សម្រាប់ជំពូក ៤ (ភាពស្របច្បាប់តាមពុទ្ធសាសនា) និងជំពូក ៣ (ស្ថាប័នពលវត្ត)។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-១៩៖** «ការតម្កល់ព្រះអដ្ឋិគូបដិបក្សក្នុងចេតិយតែមួយ» (¶6) ជាការផ្សះផ្សាក្រោយសង្គ្រាម។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-២០៖** «បណ្តាសាលើមន្ត្រីដែលធ្វេសប្រហែស» (¶12) ជាការដាក់ទោសតាមរយៈសាសនា (ជំពូក ៤)។
- **ករណីថ្មីដែលស្នើ K-ថ្មី-២១៖** «រង្វាន់ និងការលើកលែងកំណែនសម្រាប់អ្នកទាក់ដំរីស» (¶15)។

## សេចក្តីសន្និដ្ឋានក្រុម

### ករណីដែលមានសម្ភារៈគ្រប់គ្រាន់សម្រាប់គំរូវិភាគ ១០ ជំហាន (C-ids) និងមូលហេតុ
- **C06 (L022–L024, ព្រមទាំង L025)** — សម្ភារៈខ្លាំងបំផុតក្នុង G2 (S18¶6, S19¶2–¶4, S20¶1–¶39, S21¶1–¶2, S22¶1–¶4, S23¶2–¶8)។ មានការសម្រេចចិត្តជាច្រើនដែលមានហេតុផលច្បាស់ ការប្រឹក្សា ៥ ដង (S19¶2–¶4; S20¶10; S20¶25–¶26; S20¶36–¶37; S23¶25–¶29) ទំនាយ ៣ (S20¶12, ¶15, ¶22) សំបុត្រ ២ ច្បាប់ (S20¶17–¶18, ¶30–¶34) ផលវិបាកដែលអាចតាមដានបាន (ផែនដីបែកជាបី សង្គ្រាម ១០ ឆ្នាំ ការស្លាប់ដោយជំងឺ) និងទស្សនៈនៃភាគីជាច្រើន (ទាហាន ទ័ពបាសាន មន្ត្រីចតុម្មុខ)។ ត្រូវកែការពិពណ៌នានៃ C06 ឲ្យទៅជា «ជម្លោះស្នងរាជ្យបីភាគី»។
- **C07 (L025)** — S23 ទាំងមូល៖ មានព្រះរាជសារពេញ ហេតុផល (សុខុមាលភាពរាស្ត្រ ពុទ្ធសាសនា) ហេតុផលរបស់អយុធ្យាពីរកំណែ ការជួបជុំ ការចាប់ក្សត្រ រាជបណ្ណាការ ការប្រគល់គ្រួវិញ និងការលើកលែងពន្ធ។ ផលវិបាកវែង (ពញាឱងនៅអយុធ្យា ដែលនាំទៅ C13) ក៏ច្បាស់ដែរ។
- **C04 (L020)** — S17¶2–¶10៖ មានហេតុផលយុទ្ធសាស្ត្រ ៧ ចំណុច ការប្រឹក្សា ការអនុវត្ត ការបរាជ័យនៃជម្រើសដំបូង (ទឹកជំនន់នៅបាសាន) ការកែតម្រូវ និងលទ្ធផល។ ខ្សោយត្រង់កាលបរិច្ឆេទដែលមិនស៊ីគ្នា។
- **ករណីស្នើថ្មី៖ «ការចុះចូលរបស់ព្រះស្រីរាជា»** (S23¶25–¶40) អាចជាករណីស៊ីជម្រៅដាច់ដោយឡែក ឬជាផ្នែករងនៃ C06/C07។ វាមានការប្រឹក្សាដែលមានយោបល់ផ្ទុយគ្នា ហេតុផលសីលធម៌ច្បាស់ («អាណិតអស់រាស្ត្រប្រជា») និងផលវិបាកផ្ទាល់ខ្លួនដ៏ធ្ងន់ (ការចាប់ខ្លួន និងការសុគត)។

### ករណីដែលសម្ភារៈខ្សោយ
- **C05 (ដូនពេញ)**៖ មានតែ S17¶7–¶8 និង S18¶2–¶4 ដែលជារឿងនិទានប្រភពដើមឈ្មោះ (កម្រិត ឃ)។ ប៉ុន្តែ S18¶4 ពង្រឹងករណីនេះ ព្រោះព្រះរាជាចូលរួម ហើយដាក់ឈ្មោះតាមគាត់។ សម្ភារៈមានគ្រប់គ្រាន់សម្រាប់ប្រអប់ តែមិនគ្រប់សម្រាប់ការវិភាគ ១០ ជំហានពេញលេញទេ។
- **L022 ព្រះនារាយណ៍រាជា**៖ រាជ្យរបស់ព្រះអង្គមានតែពិធី និងរឿងដូនពេញ (S18)។ មិនមានការសម្រេចចិត្តនយោបាយទេ។
- **L024 ព្រះស្រីសុរិយោទ័យ**៖ ផ្នែករបស់ព្រះអង្គផ្ទាល់ (S21–S22) ខ្លីណាស់។ ទស្សនៈរបស់ព្រះអង្គមានតែតាមរយៈ S20¶13, ¶38 និង S23¶33 ដែលភាគច្រើនជាទស្សនៈរបស់អ្នកឈ្នះ។
- **K12 (រតនមាលី)**៖ មានតែក្នុង S24 ហើយមកពីសាស្ត្រាវត្តតែមួយ។ ប្រើបានសម្រាប់ការសិក្សាការចងចាំប៉ុណ្ណោះ។

### ការកែតម្រូវបញ្ជីអ្នកដឹកនាំ (ឈ្មោះថ្មី ការកែតួនាទី/ផ្នែក)
- **L020**៖ ត្រូវបន្ថែមកាលបរិច្ឆេទក្នុងតួអត្ថបទ «ព.ស. ១៩២៦ / គ.ស. ១៣៨២ (ឡើងរាជ្យ)» និងព្រះជន្ម ៥១–៧៨ រាជ្យ ៤៧ ឆ្នាំ (S17¶1, ¶12)។ មានព្រះមាតាបីអង្គនៃព្រះរាជបុត្រ (S17¶11)។ ផ្នែក៖ S17 (S14 ត្រូវផ្ទៀងផ្ទាត់ដោយក្រុមផ្សេង)។
- **L021**៖ ផ្នែកត្រូវជា «S17–S18» (មិនមែន S17 តែមួយទេ)។
- **L022**៖ ព.ស. ក្នុង S18¶1 មានកំហុស។ ព្រះជន្ម ៣៣–៣៩។ ព្រះមាតា សម្ដេចព្រះមាលាទេវី (S18¶1) ឬ អ្នកម្នាងទេវី (S17¶11)។

### ចំណុចត្រូវផ្ទៀងផ្ទាត់ជាមួយប្រភពក្រៅ
- ព្រឹត្តិការណ៍ក្នុងស្រុកសៀម (S19¶3; S20¶3, ¶5)៖ ឃុនវរវង្ស/ពញាវរវង្សា, ព្រះចៅយ៉តហ្វា, នាងស្រីសុតាចន្ទ្រ៍, ឃុនពិរេន្ទ្រទេព និងការឡើងរាជ្យរបស់ «ព្រះឌៀររាជា/ព្រះមហាចក្រពត្តិ»។ ត្រូវប្រៀបធៀបជាមួយពង្សាវតារអយុធ្យា (A004)។ ត្រូវពិនិត្យដែរថា ការរៀបរាប់ក្នុងឯកសារគោលដាក់ព្រឹត្តិការណ៍ទាំងនេះនៅ ១៤៥៣ ស៊ីនឹងប្រភពក្រៅឬទេ។
